\mel{\sharp4\;\tg{5}}, \mel{7\;\tg{1}}, \mel{\sharp1\;\tg{2}}.
On \ch{I}{}: the 5th and the root, each from under: \mel{\sharp4\;5\;7\;\hi{1}}.
\patterntab{#4 5 7 1'}

\end{patternbox}
